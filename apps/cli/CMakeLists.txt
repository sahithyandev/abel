add_library(abel_tokenizer src/tokenizer.cpp)
target_include_directories(abel_tokenizer PUBLIC src)
target_compile_features(abel_tokenizer PUBLIC cxx_std_17)

add_library(abel_parser src/parser.cpp)
target_include_directories(abel_parser PUBLIC src)
target_compile_features(abel_parser PUBLIC cxx_std_17)
target_link_libraries(abel_parser PUBLIC abel_tokenizer)

add_library(abel_evaluator src/evaluator.cpp)
target_include_directories(abel_evaluator PUBLIC src)
target_compile_features(abel_evaluator PUBLIC cxx_std_17)
target_link_libraries(abel_evaluator PUBLIC abel_parser)

add_library(abel_cli src/cli.cpp)
target_include_directories(abel_cli PUBLIC src)
target_compile_features(abel_cli PUBLIC cxx_std_17)
target_link_libraries(abel_cli PUBLIC abel_evaluator)

add_executable(abel src/main.cpp)
target_compile_features(abel PRIVATE cxx_std_17)
target_link_libraries(abel PRIVATE abel_cli)

if (MSVC)
    set(ABEL_WARNING_FLAGS /W4)
else()
    set(ABEL_WARNING_FLAGS -Wall -Wextra -Wpedantic)
endif()
foreach(target abel_tokenizer abel_parser abel_evaluator abel_cli abel)
    target_compile_options(${target} PRIVATE ${ABEL_WARNING_FLAGS})
endforeach()

if (BUILD_TESTING)
    foreach(test_name IN ITEMS tokenizer parser evaluator cli)
        add_executable(abel_${test_name}_tests tests/${test_name}_tests.cpp)
        target_compile_features(abel_${test_name}_tests PRIVATE cxx_std_17)
        target_link_libraries(abel_${test_name}_tests PRIVATE abel_${test_name})
        target_compile_options(abel_${test_name}_tests PRIVATE ${ABEL_WARNING_FLAGS})
        add_test(NAME abel_${test_name}_tests COMMAND abel_${test_name}_tests)
    endforeach()
endif()
